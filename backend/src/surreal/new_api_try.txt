pub async fn get_connection() -> Surreal<Any> {

    loop {
        match db.connect::<Ws>(env::var("SURREAL_HOST").unwrap().as_str()).await {
            Ok(client) => {
                break;
            }
            Err(e) => {
                println!("Couldn't connect to database: {:?}", e);
                sleep(Duration::from_secs(5)).await;
